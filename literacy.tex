\newcommand{\notePending}[2]{
   \begin{tcolorbox}
   [enhanced, breakable, title={#1},
      attach boxed title to top right={ xshift=-1.618em, yshift=-2mm },
      sharp corners=uphill, arc=2em,
      coltext=Silver, colframe=GhostWhite,
      colback=Black, colbacktitle=MidnightBlue,
      drop fuzzy shadow]
   #2
   \end{tcolorbox}}

\newcommand{\noteExercise}[3]{
   \begin{tcolorbox}
   [enhanced, breakable, frame hidden,
            title={#1 \small{#2}}, fonttitle=\bfseries\sffamily,
            attach boxed title to top left={ xshift=1.618em, yshift=-2.5mm },
            boxed title style={ colback=blue!50!white, drop fuzzy shadow },
            borderline north={0.5pt}{0.5pt}{MidnightBlue!50},
            coltext=Black, colback=GhostWhite, colframe=WhiteSmoke,
            sharp corners=north, arc=1em, top=1.5em]
   #3
   \end{tcolorbox}}
